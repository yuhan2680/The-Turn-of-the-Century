add_namespace = lar_fake_intel_units
country_event = {
	id = lar_fake_intel_units.1
	title = lar_fake_intel_units.1.t
	desc = lar_fake_intel_units.1.d
	picture = GFX_report_event_generic_factory
	is_triggered_only = yes
	hidden = yes
	immediate = {
		clear_temp_array = ids_to_remove
		for_each_loop = {
			array = fake_intel_ids_to_delete
			set_temp_variable = {
				passed_days = global.num_days
			}
			subtract_from_temp_variable = {
				passed_days = num_days_for_fake_intel_id@var:v
			}
			if = {
				limit = {
					check_variable = {
						passed_days > 85
					}
				}
				add_to_temp_array = {
					ids_to_remove = v
				}
				delete_unit = {
					id = v
				}
			}
		}
		for_each_loop = {
			array = ids_to_remove
			remove_from_array = {
				fake_intel_ids_to_delete = v
			}
			clear_variable = num_days_for_fake_intel_id@var:v
		}
	}
	option = {
		name = excellent
	}
}
