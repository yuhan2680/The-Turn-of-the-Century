ctc_grant_revolution_victory_rewards = {
	if = {
		limit = {
			has_country_flag = ctc_revolution_won
			has_civil_war = no
			NOT = {
				has_country_flag = ctc_revolution_rewards_granted
			}
		}
		set_country_flag = ctc_revolution_rewards_granted
		remove_ideas = ctc_prc_revolutionary_coalition
		remove_ideas = ctc_prc_army_reorganization
		add_timed_idea = {
			idea = ctc_prc_reconstruction_mobilization
			days = 1095
		}
		add_political_power = 150
		army_experience = 75
		log = "[CTC] Revolutionary victory rewards granted once, after the war."
	}
}
