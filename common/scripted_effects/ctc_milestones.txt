ctc_start_crisis = {
	if = {
		limit = {
			tag = PRC
			has_country_flag = n89_initialized
			NOT = {
				has_completed_focus = CTC_PRC_crisis
			}
		}
		complete_national_focus = CTC_PRC_crisis
	}
}
ctc_offer_conservative_ending = {
	if = {
		limit = {
			tag = PRC
			has_completed_focus = CTC_PRC_crisis
			has_country_flag = ctc_conservative_settlement_ready
			NOT = {
				has_country_flag = ctc_outcome_locked
			}
			NOT = {
				has_country_flag = ctc_outcome_pending
			}
		}
		set_country_flag = ctc_outcome_pending
		set_country_flag = ctc_pending_conservative
		country_event = {
			id = ctc_china_outcome.1
		}
	}
}
ctc_confirm_conservative_ending = {
	if = {
		limit = {
			tag = PRC
			has_completed_focus = CTC_PRC_crisis
			has_country_flag = ctc_conservative_settlement_ready
			has_country_flag = ctc_pending_conservative
			NOT = {
				has_country_flag = ctc_outcome_locked
			}
		}
		clr_country_flag = ctc_outcome_pending
		clr_country_flag = ctc_pending_conservative
		set_country_flag = ctc_outcome_locked
		set_country_flag = ctc_outcome_conservative
		complete_national_focus = CTC_PRC_stability
		ctc_apply_conservative_settlement = yes
	}
}

ctc_record_revolution_victory = {
	if = {
		limit = {
			tag = PRC
			has_completed_focus = CTC_PRC_second_revolution
			has_country_flag = ctc_outcome_revolution
			has_country_flag = ctc_revolution_defeated_government
			has_civil_war = no
			NOT = {
				has_country_flag = ctc_postwar_locked
			}
			NOT = {
				has_country_flag = ctc_postwar_pending
			}
		}
		set_country_flag = ctc_revolution_won
		set_country_flag = ctc_postwar_pending
		country_event = {
			id = ctc_china_outcome.5
		}
	}
}

		add_ideas = ctc_prc_social_democratic_settlement
		ctc_apply_social_democratic_settlement = yes
	}
}
